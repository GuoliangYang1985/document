% ===== 书名命令 =====
% 书名单点定义于此，与 _quarto.yml 的 book.title 保持一致（书名页、PDF 元数据
% 由 Quarto 从 yml 生成，页眉与封面由 \bookname 生成，两处须同步修改）。
% 用 \providecommand 以兼容"单章独立编译"（单行本 qmd 无冲突风险）。
\providecommand{\bookname}{理论测量的边界}

% ===== 抑制 Quarto 默认 \maketitle（自定义封面代行）=====
\let\maketitle\relax
\AtBeginDocument{\let\maketitle\relax}
% ===== 粗体数学 =====
% 注意：不要加载 bm 宏包！bm 会重定义 \boldsymbol，且与 unicode-math 的
% OpenType 数学字体不兼容，编译时报 "Improper alphabetic constant"。
% 统一使用 amsmath 的 \boldsymbol{X}（KaTeX/MathJax/xelatex 三方兼容）。
% ===== 中文段落：段首空两字、统一设置 =====
\setlength{\parindent}{2em}    % 段首缩进 2 个汉字宽
\setlength{\parskip}{0pt}      % 段间不留额外垂直空白（中文书传统）
\usepackage{indentfirst}       % 让 \chapter 后的第一段也跟着缩进
\usepackage[a4paper, inner=2.5cm, outer=2.0cm, top=2.5cm, bottom=2.5cm]{geometry}
\usepackage{setspace}
\setstretch{1.35}
% ===== 书籍样式页眉页脚 =====
\usepackage{fancyhdr}
\usepackage{xcolor}
\usepackage{titlesec}
\pagestyle{fancy}
\usepackage{amsthm}
% 公理/定义环境已改用 Quarto 原生定理 div（::: {#prp-*} / ::: {#def-*}），
% 其 LaTeX 环境由 Quarto 自动定义，此处不再 \newtheorem，避免重复定义冲突。
\fancyhf{}
\fancyhead[L]{\small\textit{\bookname}}
\fancyfoot[C]{\thepage}
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0pt}
\fancypagestyle{plain}{
    \fancyhf{}
    \fancyfoot[C]{\thepage}
    \renewcommand{\headrulewidth}{0pt}
    \renewcommand{\footrulewidth}{0pt}
}
% ===== 章节标题美化 =====
% 强制全局使用阿拉伯数字编号（第1章）
\CTEXsetup[
    name={第,章},
    number=\arabic{chapter}
]{chapter}
% 章号自动改为"第N章"（ctexbook 内置），并按 display 样式居中。
\titleformat{\chapter}
    {\Huge\bfseries\color{blue!50!black}\centering}
    {第\thechapter 章}{1em}{}

\titleformat{\section}
    {\Large\bfseries\color{blue!40!black}}
    {\thesection}{1em}{}

\titleformat{\subsection}
    {\large\bfseries\color{blue!30!black}}
    {\thesubsection}{0.8em}{}

\titleformat{\subsubsection}
    {\normalsize\bfseries\color{blue!20!black}}
    {\thesubsubsection}{0.6em}{}

% ===== 页码体系：全书连续阿拉伯数字，目录页码 = PDF 物理页码 =====
% ctexbook 的 \frontmatter / \mainmatter / \backmatter 除了切换 \@mainmatter
% 语义（决定 \chapter 是否编号），还会切换页码体系（roman ↔ arabic）并把
% page 计数器清零。Quarto 会在正文第一章之前额外插入一次 \mainmatter，
% 于是清零发生两次：目录里「前言」与「第 1 章」各自从 1 开始，出现两组
% 不同的第 1 页，读者按目录数字无法定位。
% 此处保留三者的 \@mainmatter 语义与 \cleardoublepage，去掉页码体系切换
% 与清零，使目录中的数字与 PDF 阅读器的跳转页号逐页对齐。
\makeatletter
\renewcommand{\frontmatter}{\cleardoublepage\@mainmatterfalse}
\renewcommand{\mainmatter}{\cleardoublepage\@mainmattertrue}
\renewcommand{\backmatter}{\cleardoublepage\@mainmatterfalse}
\makeatother

% ===== 5、6 级标题（段落头 + runin）：必须补齐，
%        否则一旦文档里出现 ##### ###### 就会报
%        "Package titlesec Error: No format for this command." =====
\titleformat{\paragraph}[runin]
    {\normalsize\bfseries}{}{0pt}{}
\titleformat{\subparagraph}[runin]
    {\normalsize\itshape}{}{0pt}{}

% ===== 图索引 / 表索引 进目录 =====
% \listoffigures 与 \listoftables 生成的是无编号章（chapter*），默认不写目录，
% 于是目录里能看到「符号与约定」「关键词索引」「参考文献」，却看不到实际存在的
% 图索引与表索引。用 etoolbox 在这两条命令执行前补写一条目录项。
\usepackage{etoolbox}
\pretocmd{\listoffigures}{\phantomsection\addcontentsline{toc}{chapter}{图索引}}{}{}
\pretocmd{\listoftables}{\phantomsection\addcontentsline{toc}{chapter}{表索引}}{}{}
